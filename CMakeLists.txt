cmake_minimum_required(VERSION 3.16)
project(aino_player VERSION 0.3.0 LANGUAGES CXX C)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

find_package(Threads REQUIRED)

add_library(player_desktop STATIC src/desktop.cpp)
target_link_libraries(player_desktop PUBLIC Threads::Threads)
if(WIN32)
  if(NOT MSVC)
    message(FATAL_ERROR "Windows desktop media controls require MSVC and Windows SDK with C++/WinRT; build the FFmpeg helper separately with GCC")
  endif()
  # 新 Windows SDK 的 C++/WinRT 使用标准 coroutine；其余模块仍为 C++17。
  target_compile_features(player_desktop PRIVATE cxx_std_20)
  target_link_libraries(player_desktop PUBLIC user32 shell32 advapi32 runtimeobject windowsapp iphlpapi ws2_32)
else()
  find_package(X11 REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(DBUS REQUIRED IMPORTED_TARGET dbus-1)
  target_include_directories(player_desktop PRIVATE ${X11_INCLUDE_DIR})
  target_link_libraries(player_desktop PUBLIC PkgConfig::DBUS ${X11_LIBRARIES})
endif()

add_library(player_audio STATIC src/audio.cpp)
target_link_libraries(player_audio PUBLIC Threads::Threads ${CMAKE_DL_LIBS})

option(AINO_BUNDLE_FFMPEG "Embed the minimal local audio decoder (recommended for distribution)" ON)
set(AINO_FFMPEG_EXECUTABLE "" CACHE FILEPATH "Minimal FFmpeg helper to embed in the player")
if(AINO_BUNDLE_FFMPEG AND NOT AINO_FFMPEG_EXECUTABLE)
  if(WIN32 AND EXISTS "${CMAKE_SOURCE_DIR}/build/codec/ffmpeg.exe")
    set(AINO_FFMPEG_EXECUTABLE "${CMAKE_SOURCE_DIR}/build/codec/ffmpeg.exe")
  elseif(NOT WIN32 AND EXISTS "${CMAKE_SOURCE_DIR}/build/codec/ffmpeg")
    set(AINO_FFMPEG_EXECUTABLE "${CMAKE_SOURCE_DIR}/build/codec/ffmpeg")
  endif()
endif()
if(AINO_BUNDLE_FFMPEG)
  if(NOT AINO_FFMPEG_EXECUTABLE)
    message(FATAL_ERROR "Build the decoder first: python tools/build_ffmpeg.py; or set AINO_FFMPEG_EXECUTABLE. For a developer-only build, use -DAINO_BUNDLE_FFMPEG=OFF")
  endif()
  if(NOT EXISTS "${AINO_FFMPEG_EXECUTABLE}")
    message(FATAL_ERROR "AINO_FFMPEG_EXECUTABLE does not exist: ${AINO_FFMPEG_EXECUTABLE}")
  endif()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(codec_header "${CMAKE_BINARY_DIR}/generated/aino_ffmpeg_data.hpp")
  add_custom_command(OUTPUT "${codec_header}"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/build_ffmpeg.py"
      --embed "${AINO_FFMPEG_EXECUTABLE}" --header "${codec_header}"
    DEPENDS "${AINO_FFMPEG_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/build_ffmpeg.py" "${CMAKE_SOURCE_DIR}/vendor/LICENSE.FFmpeg.txt" VERBATIM)
  target_sources(player_audio PRIVATE "${codec_header}")
  target_include_directories(player_audio PRIVATE "${CMAKE_BINARY_DIR}/generated")
  target_compile_definitions(player_audio PRIVATE AINO_BUNDLED_FFMPEG=1)
  # Embed raw bytes without parsing millions of C++ integer literals.
  file(TO_CMAKE_PATH "${AINO_FFMPEG_EXECUTABLE}" codec_binary)
  if(WIN32)
    enable_language(RC)
    set(codec_resource "${CMAKE_BINARY_DIR}/generated/aino_ffmpeg.rc")
    file(GENERATE OUTPUT "${codec_resource}" CONTENT "32761 RCDATA \"${codec_binary}\"\n")
  else()
    enable_language(ASM)
    set(codec_resource "${CMAKE_BINARY_DIR}/generated/aino_ffmpeg.S")
    file(GENERATE OUTPUT "${codec_resource}" CONTENT ".section .rodata\n.balign 16\n.global aino_ffmpeg_start\n.global aino_ffmpeg_end\naino_ffmpeg_start:\n.incbin \"${codec_binary}\"\naino_ffmpeg_end:\n.section .note.GNU-stack,\"\"\n")
    target_sources(player_audio PRIVATE "${codec_resource}")
  endif()
  set_source_files_properties("${codec_resource}" PROPERTIES OBJECT_DEPENDS "${AINO_FFMPEG_EXECUTABLE}")
else()
  message(STATUS "Bundled decoder disabled; this developer build requires external FFmpeg for AAC/M4A")
endif()

function(aino_attach_codec target)
  if(AINO_BUNDLE_FFMPEG AND WIN32)
    target_sources(${target} PRIVATE "${codec_resource}")
  endif()
endfunction()

add_executable(aino_player WIN32 src/main.cpp src/cover_decode.cpp)
target_compile_definitions(aino_player PRIVATE AINO_PLAYER_BUSINESS_UNIT=1)
option(AINO_PLAYER_PCH "Cache stable framework parsing for incremental player builds" ON)
if(AINO_PLAYER_PCH)
  # 框架公开实现全部 inline，沿用原单元和宏，仅缓存稳定的解析结果，不改 ABI / 上游。
  file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/generated/aino_pch.hpp" CONTENT
    "#define AINO_IMPLEMENTATION\n#include \"${CMAKE_CURRENT_SOURCE_DIR}/src/aino.hpp\"\n#ifdef AINO_PLATFORM_X11\n#undef Bool\n#undef Status\n#undef None\n#endif\n")
  target_precompile_headers(aino_player PRIVATE "${CMAKE_BINARY_DIR}/generated/aino_pch.hpp")
  set_source_files_properties(src/cover_decode.cpp PROPERTIES SKIP_PRECOMPILE_HEADERS ON)
endif()
# JS 是目录接口的唯一实现。内嵌默认脚本，仍允许 providers/ 同名文件覆盖；
# 单 EXE 和 Linux 安装均不依赖启动工作目录。
file(GLOB platform_scripts CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/providers/*.js")
set(platform_header "#pragma once\nnamespace prov { inline const std::pair<const char*, const char*> bundled_scripts[] = {\n")
foreach(script IN LISTS platform_scripts)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${script}")
  get_filename_component(script_name "${script}" NAME)
  file(READ "${script}" script_code)
  string(APPEND platform_header "{\"${script_name}\", R\"AINOPLATFORM(${script_code})AINOPLATFORM\"},\n")
endforeach()
string(APPEND platform_header "}; }\n")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/generated/aino_platforms.hpp" CONTENT "${platform_header}")
target_include_directories(aino_player PRIVATE "${CMAKE_BINARY_DIR}/generated")
aino_attach_codec(aino_player)
target_link_libraries(aino_player PRIVATE player_audio player_login player_desktop quickjs)
if(WIN32)
  enable_language(RC)
  configure_file(packaging/windows.rc.in generated/aino_player.rc @ONLY)
  target_sources(aino_player PRIVATE "${CMAKE_BINARY_DIR}/generated/aino_player.rc")
  target_link_libraries(aino_player PRIVATE user32 gdi32 winhttp shell32 ole32 windowscodecs advapi32 bcrypt)
  if(MSVC)
    target_link_options(aino_player PRIVATE /ENTRY:mainCRTStartup)
  else()
    target_link_options(aino_player PRIVATE -static -mwindows)
  endif()
else()
  find_package(X11 REQUIRED)
  target_include_directories(aino_player PRIVATE ${X11_INCLUDE_DIR})
  target_link_libraries(aino_player PRIVATE ${X11_LIBRARIES} CURL::libcurl ${CMAKE_DL_LIBS})
endif()

foreach(target player_audio player_login player_desktop aino_player)
  target_compile_definitions(${target} PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN AINO_PLAYER_VERSION="${PROJECT_VERSION}")
  # sources.hpp → lxsource.hpp → quickjs.h
  target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/vendor/quickjs)
  if(MSVC)
    target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/utf-8> $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
  endif()
endforeach()

# 发布版优先紧凑代码，允许链接器回收独立函数和数据；不压缩或运行时解包主程序。
if(MSVC)
  foreach(target player_audio player_login player_desktop quickjs aino_player)
    target_compile_options(${target} PRIVATE "$<$<AND:$<CONFIG:Release>,$<COMPILE_LANGUAGE:C,CXX>>:/Os;/Gy;/Gw>")
  endforeach()
  target_link_options(aino_player PRIVATE $<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF>)
endif()

# 测试程序也会 include 到 business.hpp / sources.hpp 这条链（→ quickjs.h），
# 所以必须给它们同样的头文件路径，并链接 quickjs（否则链接期 Engine 符号缺失）。
function(aino_link_quickjs target)
  target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/vendor/quickjs "${CMAKE_BINARY_DIR}/generated")
  target_link_libraries(${target} PRIVATE quickjs)
endfunction()

  elseif(WIN32)

  endif()

    find_program(AINO_TEST_BROWSER NAMES chromium chromium-browser google-chrome chrome.exe msedge.exe

    add_executable(controller_tests tests/controller_tests.cpp src/cover_decode.cpp)
